# retina-iris-pds crontab
#
# Log paths must match LOG_DIR in conf/tier1_settings.conf / pd_settings.conf;
# cron can't source those files, so keep them in sync by hand.
#
# cron's default PATH is minimal, hence the explicit line. gcloud must be on it:
# the wrappers fetch BGP_API_KEYS and IRIS_PASSWORD from Secret Manager.
PATH=/opt/retina/bin:/usr/local/bin:/usr/bin:/bin

# Uses today's 00:00 UTC RIB snapshot; takes ~4h.
0 8 * * *	/opt/retina/retina-iris-pds/cron/tier1_wrapper.sh >> /opt/retina/logs/tier1_wrapper.log 2>&1

# Previous day's PDs; runs after the last Iris run has landed, before the 04:00 run.
0 2 * * *	/opt/retina/retina-iris-pds/cron/pd_wrapper.sh >> /opt/retina/logs/pd_wrapper.log 2>&1
